cmake_minimum_required(VERSION 3.27)
project(ane_linear LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Python 3.10 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE NB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
list(APPEND CMAKE_PREFIX_PATH "${NB_DIR}")
find_package(nanobind CONFIG REQUIRED)
find_package(MLX CONFIG REQUIRED)
# The nanobind ABI must match the one MLX was built with, or MLX arrays are rejected at run time.
#   MLX 0.32.0 to 0.32.3: ABI v20 (nanobind 2.13.x or 2.14.x)
#   MLX 0.32.4 and later: ABI v22 (nanobind 3.1.x)
if(MLX_VERSION VERSION_GREATER_EQUAL 0.32.4)
  if(NOT nanobind_VERSION VERSION_GREATER_EQUAL 3.1.0 OR nanobind_VERSION VERSION_GREATER_EQUAL 3.2.0)
    message(FATAL_ERROR "MLX ${MLX_VERSION} needs nanobind 3.1.x. Found ${nanobind_VERSION}.")
  endif()
else()
  if(NOT nanobind_VERSION VERSION_GREATER_EQUAL 2.13.0 OR nanobind_VERSION VERSION_GREATER_EQUAL 2.15.0)
    message(FATAL_ERROR "MLX ${MLX_VERSION} needs nanobind 2.13.x or 2.14.x. Found ${nanobind_VERSION}.")
  endif()
endif()
nanobind_add_module(_ane_linear NB_STATIC NB_DOMAIN mlx STABLE_ABI ane_linear.cpp ane_bridge.mm)
target_link_libraries(_ane_linear PRIVATE mlx "-framework CoreML" "-framework Foundation")
set_source_files_properties(ane_bridge.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
